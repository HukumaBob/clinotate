# Video Review Player Layout Redesign + Crop UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Remove the redundant page-level Media heading, restructure the
video review player into a bounded-width video + contextual side panel
layout, and make the crop box a first-class, visible, editable part of
both sequence creation and re-cut (with a live overlay and inheritance
from the video's most recent sequence).

**Architecture:** A new presentational `CropOverlay` component renders
percentage-based inset bars over a `position: relative` wrapper around
the `<video>` element. `VideoReviewPlayer` restructures its JSX into a
flex row (video left, contextual side panel right) and moves the
Mark-start/Mark-end buttons out of `SequenceTimeline` into that side
panel. The backend's `SequenceUpdate` schema and `recut_sequence` gain an
optional `crop` parameter that, when provided, is persisted as the new
`crop_box`; when omitted, today's fallback (reuse the existing
`crop_box`) is preserved.

**Tech Stack:** FastAPI + SQLAlchemy + Pydantic (backend), React +
TanStack Query + Tailwind (frontend), pytest + vitest/testing-library
(tests).

```python
    Extracts and uploads the new frames first, then deletes the old
    Frame rows (cascading to their Annotation rows) and updates the
    sequence's timing -- in that order, so a failure during extraction
```

```typescript
export function useUpdateSequence() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (params: {
      sequenceId: number;
      name?: string;
      procedureId?: number;
      startTime?: number;
      duration?: number;
      fps?: number;
      crop?: { left: number; right: number; top: number; bottom: number };
    }) => {
      const result = await apiClient.PATCH("/media/sequences/{sequence_id}", {
        params: { path: { sequence_id: params.sequenceId } },
        body: {
          name: params.name,
          procedure_id: params.procedureId,
          start_time: params.startTime,
          duration: params.duration,
          fps: params.fps,
          crop: params.crop,
        },
      });
      if (result.error) throw result.error;
      return result.data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["sequences"] });
      queryClient.invalidateQueries({ queryKey: ["video-sequences"] });
      queryClient.invalidateQueries({ queryKey: ["media-library"] });
    },
  });
}
```

- Produces: `RecutSequenceForm` gains a new required prop `videoSrc:

**Important design correction from the spec:** the spec described
`CropOverlay` as layered over "the video" inside `VideoReviewPlayer`'s
existing single `<video>` element, shared across whichever form is
active. Rendering a full second `<video>` inside `RecutSequenceForm` and
`SequenceFromMarksForm` each time would duplicate playback state and be
wasteful. Instead: `RecutSequenceForm` and `SequenceFromMarksForm` each
render their OWN lightweight `<img>`-based static preview (not a second
`<video>` element) using the video's current display frame — simplest
correct approach is a static preview using the SAME `stream_url` in a
paused, muted, non-interactive `<video>` sized identically via
`className="pointer-events-none max-h-[40vh] w-full rounded bg-black"` (matching
`VideoReviewPlayer`'s main video's existing `max-h-[40vh]` cap so the
preview visually matches). This keeps each form self-contained and
testable without threading a shared video ref through props. Pass
`videoSrc: string` for this purpose.

```typescript
import { useState } from "react";

import { useUpdateSequence } from "../../entities/useSequences";
import type { components } from "../../shared/api/generated/schema";
import { CropOverlay } from "./CropOverlay";

type SequenceRead = components["schemas"]["SequenceRead"];
type Crop = { left: number; right: number; top: number; bottom: number };

const ZERO_CROP: Crop = { left: 0, right: 0, top: 0, bottom: 0 };

export function RecutSequenceForm({
  sequence,
  markStart,
  markEnd,
  videoSrc,
  onDone,
}: {
  sequence: SequenceRead;
  markStart: number;
  markEnd: number;
  videoSrc: string;
  onDone: () => void;
}) {
  const [fps, setFps] = useState<number | "">("");
  const [crop, setCrop] = useState<Crop>(
    (sequence.crop_box as Crop | null) ?? ZERO_CROP
  );
  const [confirming, setConfirming] = useState(false);
  const updateSequence = useUpdateSequence();

  const startTime = Math.round(Math.min(markStart, markEnd));
  const duration = Math.round(Math.abs(markEnd - markStart));

  function handleRecutClick(event: React.FormEvent) {
    event.preventDefault();
    setConfirming(true);
  }

  function handleConfirm() {
    if (fps === "") return;
    updateSequence.mutate(
      { sequenceId: sequence.id, startTime, duration, fps, crop },
      {
        onSuccess: () => onDone(),
        onError: () => setConfirming(false),
      }
    );
  }

  return (
    <form onSubmit={handleRecutClick} className="flex flex-col gap-3 rounded border border-gray-800 bg-gray-900 p-4">
      <div className="relative">
        <video src={videoSrc} muted className="pointer-events-none max-h-[40vh] w-full rounded bg-black" />
        <CropOverlay crop={crop} />
      </div>
      <p className="text-sm text-gray-400">
        New range: {startTime}s – {startTime + duration}s ({duration}s)
      </p>
      <label className="flex w-24 flex-col gap-1 text-sm text-gray-300">
        FPS
        <input
          type="number"
          value={fps}
          onChange={(e) => setFps(e.target.value === "" ? "" : Number(e.target.value))}
          className="rounded border border-gray-700 bg-gray-800 px-3 py-2 text-gray-100"
          required
        />
      </label>
      <div className="flex gap-3">
        {(["left", "right", "top", "bottom"] as const).map((side) => (
          <label key={side} className="flex flex-col gap-1 text-sm text-gray-300">
            {side} %
            <input
              type="number"
              value={crop[side]}
              onChange={(e) => setCrop((c) => ({ ...c, [side]: Number(e.target.value) }))}
              className="w-20 rounded border border-gray-700 bg-gray-800 px-3 py-2 text-gray-100"
            />
          </label>
        ))}
      </div>
      {sequence.annotation_count > 0 && (
        <p className="text-sm text-red-400">
          This will delete {sequence.annotation_count} annotation{sequence.annotation_count === 1 ? "" : "s"} on this
          sequence&apos;s current frames.
        </p>
      )}
      {updateSequence.isError && (
        <p className="text-sm text-red-400">Re-cut failed. Please try again.</p>
      )}
      {!confirming ? (
        <button
          type="submit"
          disabled={fps === ""}
          className="rounded bg-blue-600 px-3 py-2 font-medium text-white hover:bg-blue-500 disabled:opacity-50"
        >
          Re-cut
        </button>
      ) : (
        <div className="flex gap-2">
          <button
            type="button"
            onClick={handleConfirm}
            disabled={updateSequence.isPending}
            className="rounded bg-red-600 px-3 py-2 font-medium text-white hover:bg-red-500 disabled:opacity-50"
          >
            Confirm
          </button>
          <button
            type="button"
            onClick={() => setConfirming(false)}
            className="rounded border border-gray-700 px-3 py-2 text-sm hover:bg-gray-800"
          >
            Cancel
          </button>
        </div>
      )}
    </form>
  );
}
```

```typescript
import { useProcedures } from "../../entities/useProcedures";
import { useCreateSequenceFromVideo } from "../../entities/useSequences";

type Crop = { left: number; right: number; top: number; bottom: number };

const ZERO_CROP: Crop = { left: 0, right: 0, top: 0, bottom: 0 };

function mostRecentCrop(sequences: SequenceRead[]): Crop {
  const mostRecent = [...sequences].sort((a, b) => b.id - a.id)[0];
  return (mostRecent?.crop_box as Crop | null | undefined) ?? ZERO_CROP;
}

export function SequenceFromMarksForm({
  videoId,
  markStart,
  markEnd,
  procedureId: procedureIdProp,
  videoSrc,
  existingSequences,
  onCreated,

    createSequence.mutate(
```

```typescript
import { useRef, useState } from "react";

import { useVideoSequences } from "../../entities/useSequences";
import { useVideo } from "../../entities/useVideo";
import { RecutSequenceForm } from "./RecutSequenceForm";
import { RenameSequenceForm } from "./RenameSequenceForm";
import { SequenceFromMarksForm } from "./SequenceFromMarksForm";
import { SequenceTimeline } from "./SequenceTimeline";
import { TranscodePrompt } from "./TranscodePrompt";

export function VideoReviewPlayer({
  videoId,
  onBack,
  onOpenSequence,
}: {
  videoId: number;
  onBack: () => void;
  onOpenSequence: (sequenceId: number) => void;
}) {
  const videoQuery = useVideo(videoId);
  const sequencesQuery = useVideoSequences(videoId);
  const videoRef = useRef<HTMLVideoElement>(null);
  const [markStart, setMarkStart] = useState<number | null>(null);
  const [markEnd, setMarkEnd] = useState<number | null>(null);
  const [videoReady, setVideoReady] = useState(false);
  const [renamingSequenceId, setRenamingSequenceId] = useState<number | null>(null);
  const [recuttingSequenceId, setRecuttingSequenceId] = useState<number | null>(null);

  function handleMarkStart() {
    if (videoRef.current) setMarkStart(videoRef.current.currentTime);
  }

  function handleMarkEnd() {
    if (videoRef.current) setMarkEnd(videoRef.current.currentTime);
  }

  function handleRenameSequence(sequenceId: number) {
    setRecuttingSequenceId(null);
    setRenamingSequenceId(sequenceId);
  }

  function handleRecutSequence(sequenceId: number) {
    setRenamingSequenceId(null);
    setRecuttingSequenceId(sequenceId);
    const target = sequencesQuery.data?.find((s) => s.id === sequenceId);
    if (target) {
      setMarkStart(target.start_time);
      setMarkEnd(target.start_time + target.duration);
    }
  }

  function handleEditDone() {
    setRenamingSequenceId(null);
    setRecuttingSequenceId(null);
    setMarkStart(null);
    setMarkEnd(null);
    sequencesQuery.refetch();
  }

  function handleCreated() {
    setMarkStart(null);
    setMarkEnd(null);
    sequencesQuery.refetch();
  }

  const video = videoQuery.data;

  return (
    <div className="flex flex-col gap-4 rounded border border-gray-800 bg-gray-900 p-4">
      <div className="flex items-center justify-between">
        <h2 className="text-lg font-medium text-gray-100">{video?.title ?? "Video"}</h2>
        <button type="button" onClick={onBack} className="text-sm text-blue-400 hover:underline">
          ← Upload another video
        </button>
      </div>

      {video?.browser_playable === false ? (
        <TranscodePrompt videoId={videoId} onTranscodeComplete={() => videoQuery.refetch()} />
      ) : video ? (
        <>
          <div className="flex flex-col gap-4 lg:flex-row">
            <video
              ref={videoRef}
              src={video.stream_url}
              controls
              className="max-h-[40vh] flex-[3] rounded bg-black"
              onLoadedMetadata={() => setVideoReady(true)}
            />
            <div className="flex flex-[2] flex-col gap-3 rounded border border-gray-800 bg-gray-950 p-3">
              <div className="flex gap-3">
                <button
                  type="button"
                  onClick={handleMarkStart}
                  disabled={!videoReady}
                  className="rounded border border-gray-700 px-3 py-1 text-sm hover:bg-gray-800 disabled:opacity-50"
                >
                  Mark start
                </button>
                <button
                  type="button"
                  onClick={handleMarkEnd}
                  disabled={!videoReady}
                  className="rounded border border-gray-700 px-3 py-1 text-sm hover:bg-gray-800 disabled:opacity-50"
                >
                  Mark end
                </button>
              </div>
              {renamingSequenceId !== null && (() => {
                const target = sequencesQuery.data?.find((s) => s.id === renamingSequenceId);
                return target ? (
                  <RenameSequenceForm sequence={target} onDone={handleEditDone} />
                ) : null;
              })()}
              {recuttingSequenceId !== null && markStart !== null && markEnd !== null && (() => {
                const target = sequencesQuery.data?.find((s) => s.id === recuttingSequenceId);
                return target ? (
                  <RecutSequenceForm
                    sequence={target}
                    markStart={markStart}
                    markEnd={markEnd}
                    videoSrc={video.stream_url}
                    onDone={handleEditDone}
                  />
                ) : null;
              })()}
              {recuttingSequenceId === null && renamingSequenceId === null && markStart !== null && markEnd !== null && markEnd !== markStart && (
                <SequenceFromMarksForm
                  videoId={videoId}
                  markStart={markStart}
                  markEnd={markEnd}
                  procedureId={video.procedure_id ?? undefined}
                  videoSrc={video.stream_url}
                  existingSequences={sequencesQuery.data ?? []}
                  onCreated={handleCreated}
                />
              )}
              {recuttingSequenceId === null && renamingSequenceId === null && markStart !== null && markEnd !== null && markEnd === markStart && (
                <p className="text-sm text-red-400">Start and end can&apos;t be the same point.</p>
              )}
            </div>
          </div>
          <SequenceTimeline
            durationSeconds={video.duration_seconds}
            sequences={sequencesQuery.data ?? []}
            markStart={markStart}
            markEnd={markEnd}
            onOpenSequence={onOpenSequence}
            onRenameSequence={handleRenameSequence}
            onRecutSequence={handleRecutSequence}
          />
        </>
      ) : null}
    </div>
  );
}
```
