# Walking Skeleton Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the new endoscopy annotation platform's foundational skeleton — infrastructure, auth, taxonomy, media ingestion, and manual classification annotation — as working, end-to-end, testable software, with no ML inference yet.

**Architecture:** FastAPI modular monolith (`auth`, `taxonomy`, `media`, `annotation` packages) backed by PostgreSQL, with file bytes in MinIO (S3 API) behind a `StorageService` abstraction. React (Vite) + TypeScript SPA with Tailwind (dark theme default), TanStack Query for server state, Zustand reserved for later UI-state needs. Docker Compose orchestrates Postgres, Redis (provisioned now, unused until the ML worker plan), and MinIO. No Celery worker and no ML adapters in this plan — those are a follow-up plan.

**Tech Stack:** Python 3.12, FastAPI, SQLAlchemy 2.0 + Alembic, `uv` for dependency management, pytest, PostgreSQL 16, MinIO, Node 20, React 18 + Vite, TypeScript, Tailwind CSS, TanStack Query, `openapi-typescript` for generated API client, Vitest + React Testing Library, Playwright for one end-to-end smoke test.

## Global Constraints

- License: Apache 2.0. `LICENSE` and `NOTICE` files must exist at repo root before any other file is committed.
- Language: English only in all code, comments, docstrings, README, and UI copy. No Russian in the repository.
- UI theme: dark is the default and only implemented theme; Tailwind config must use the `class` dark-mode strategy so light mode can be added later without reworking components.
- Repository location: new directory `<repo-root>`, a fresh git repository, independent of `segmentation`.
- Object storage: all binary media (video, frames) goes through a `StorageService` abstraction backed by MinIO (S3 API) — never direct filesystem paths in domain code.
- `taxonomy_version` must be attached to any data that references a `Label`, per the design spec — annotations store the taxonomy version they were created against.
- API contract: REST, OpenAPI schema auto-generated by FastAPI, TypeScript client/types generated from that schema — never hand-written fetch calls with untyped payloads.
- TDD: every backend task and every frontend task with logic writes a failing test before the implementation.

- [ ] **Step 3: Write storage.py**

```python
from functools import lru_cache

import boto3
from botocore.client import Config
from botocore.exceptions import ClientError

from app.config import Settings, get_settings


class StorageService:
    def __init__(self, settings: Settings):
        self._bucket = settings.minio_bucket
        protocol = "https" if settings.minio_secure else "http"
        self._client = boto3.client(
            "s3",
            endpoint_url=f"{protocol}://{settings.minio_endpoint}",
            aws_access_key_id=settings.minio_access_key,
            aws_secret_access_key=settings.minio_secret_key,
            config=Config(signature_version="s3v4"),
        )

    def ensure_bucket(self) -> None:
        try:
            self._client.head_bucket(Bucket=self._bucket)
        except ClientError:
            self._client.create_bucket(Bucket=self._bucket)

    def upload(self, key: str, data: bytes, content_type: str) -> None:
        self._client.put_object(
            Bucket=self._bucket, Key=key, Body=data, ContentType=content_type
        )

    def download(self, key: str) -> bytes:
        response = self._client.get_object(Bucket=self._bucket, Key=key)
        return response["Body"].read()

    def delete(self, key: str) -> None:
        self._client.delete_object(Bucket=self._bucket, Key=key)

    def url_for(self, key: str) -> str:
        return self._client.generate_presigned_url(
            "get_object",
            Params={"Bucket": self._bucket, "Key": key},
            ExpiresIn=3600,
        )


@lru_cache
def get_storage_service() -> StorageService:
    service = StorageService(get_settings())
    service.ensure_bucket()
    return service
```

from passlib.context import CryptContext

from app.config import get_settings

_pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def hash_password(password: str) -> str:
    return _pwd_context.hash(password)


def verify_password(password: str, hashed: str) -> bool:
    return _pwd_context.verify(password, hashed)


def create_access_token(user_id: int) -> str:
    settings = get_settings()
    expire = datetime.now(timezone.utc) + timedelta(minutes=settings.jwt_expire_minutes)
    payload = {"sub": str(user_id), "exp": expire}
    return jwt.encode(payload, settings.jwt_secret, algorithm=settings.jwt_algorithm)


def decode_access_token(token: str) -> int:
    settings = get_settings()
    try:
        payload = jwt.decode(token, settings.jwt_secret, algorithms=[settings.jwt_algorithm])
    except JWTError as exc:
        raise ValueError("invalid or expired token") from exc
    return int(payload["sub"])

    user = User(email=payload.email, hashed_password=hash_password(payload.password), role=payload.role)
    session.add(user)
    try:
        session.commit()
    except IntegrityError as exc:
        session.rollback()
        raise HTTPException(status_code=status.HTTP_409_CONFLICT, detail="Email already registered") from exc
    session.refresh(user)
    return user


@router.post("/login", response_model=TokenResponse)
def login(payload: UserLogin, session: Session = Depends(get_session)) -> TokenResponse:
    user = session.query(User).filter(User.email == payload.email).first()
    if user is None or not verify_password(payload.password, user.hashed_password):
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid credentials")
    return TokenResponse(access_token=create_access_token(user.id))

from sqlalchemy import text
from sqlalchemy.orm import Session

from app.taxonomy.models import Label


def get_label_descendants(session: Session, label_id: int) -> list[Label]:
    query = text(
        """
        WITH RECURSIVE descendants AS (
            SELECT child_id AS label_id FROM label_relations WHERE parent_id = :label_id
            UNION
            SELECT lr.child_id FROM label_relations lr
            INNER JOIN descendants d ON lr.parent_id = d.label_id
        )
        SELECT label_id FROM descendants
        """
    )
    rows = session.execute(query, {"label_id": label_id}).fetchall()
    ids = [row.label_id for row in rows]
    if not ids:
        return []
    return session.query(Label).filter(Label.id.in_(ids)).all()
```

import cv2


def compute_checksum(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def probe_video(video_bytes: bytes) -> tuple[float, float]:
    """Returns (fps, duration_seconds)."""
    with tempfile.NamedTemporaryFile(suffix=".mp4") as tmp:
        tmp.write(video_bytes)
        tmp.flush()
        capture = cv2.VideoCapture(tmp.name)
        fps = capture.get(cv2.CAP_PROP_FPS) or 0.0
        frame_count = capture.get(cv2.CAP_PROP_FRAME_COUNT) or 0.0
        capture.release()
    duration = frame_count / fps if fps > 0 else 0.0
    return fps, duration


def extract_frames(video_bytes: bytes, sample_fps: float = 1.0) -> list[tuple[int, bytes, int, int]]:
    """Extracts frames at `sample_fps`, returns (frame_index, png_bytes, width, height)."""
    results: list[tuple[int, bytes, int, int]] = []
    with tempfile.NamedTemporaryFile(suffix=".mp4") as tmp:
        tmp.write(video_bytes)
        tmp.flush()
        capture = cv2.VideoCapture(tmp.name)
        source_fps = capture.get(cv2.CAP_PROP_FPS) or sample_fps
        step = max(int(round(source_fps / sample_fps)), 1)

        frame_index = 0
        source_frame_number = 0
        while True:
            success, frame = capture.read()
            if not success:
                break
            if source_frame_number % step == 0:
                height, width = frame.shape[:2]
                success_encode, buffer = cv2.imencode(".png", frame)
                if success_encode:
                    results.append((frame_index, buffer.tobytes(), width, height))
                    frame_index += 1
            source_frame_number += 1
        capture.release()
    return results
```

@router.post("/videos", response_model=VideoRead, status_code=status.HTTP_201_CREATED)
async def upload_video(
    file: UploadFile,
    title: str,
    session: Session = Depends(get_session),
    storage: StorageService = Depends(get_storage_service),
) -> Video:
    data = await file.read()
    fps, duration = probe_video(data)
    checksum = compute_checksum(data)
    storage_key = f"videos/{checksum}.mp4"
    storage.upload(storage_key, data, content_type="video/mp4")

    video = Video(storage_key=storage_key, title=title, fps=fps, duration_seconds=duration, checksum=checksum)
    session.add(video)
    session.commit()
    session.refresh(video)
    return video

@router.post("/videos/{video_id}/sequences", response_model=SequenceRead, status_code=status.HTTP_201_CREATED)
def create_sequence(
    video_id: int,
    payload: SequenceCreate,
    session: Session = Depends(get_session),
    storage: StorageService = Depends(get_storage_service),
) -> Sequence:
    video = session.get(Video, video_id)
    if video is None:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Video not found")

    sequence = Sequence(video_id=video_id, start_time=payload.start_time, duration=payload.duration)
    session.add(sequence)
    session.commit()
    session.refresh(sequence)

    video_bytes = storage.download(video.storage_key)
    extracted = extract_frames(video_bytes, sample_fps=1.0)
    for frame_index, png_bytes, width, height in extracted:
        storage_key = f"frames/{sequence.id}/{frame_index}.png"
        storage.upload(storage_key, png_bytes, content_type="image/png")
        session.add(
            Frame(
                sequence_id=sequence.id,
                storage_key=storage_key,
                frame_index=frame_index,
                width=width,
                height=height,
            )
        )
    session.commit()
    return sequence

```typescript
import { create } from "zustand";

import { apiClient, setAuthToken } from "../../shared/api/client";

const TOKEN_STORAGE_KEY = "endoscopy_auth_token";

interface AuthUser {
  id: number;
  email: string;
  role: "annotator" | "reviewer" | "admin";
}

interface AuthState {
  token: string | null;
  user: AuthUser | null;
  isAuthenticated: boolean;
  login: (email: string, password: string) => Promise<void>;
  logout: () => void;
}

export const useAuth = create<AuthState>((set) => ({
  token: localStorage.getItem(TOKEN_STORAGE_KEY),
  user: null,
  isAuthenticated: false,

  login: async (email: string, password: string) => {
    const loginResult = await apiClient.POST("/auth/login", {
      body: { email, password },
    });
    if (!loginResult.data) {
      throw new Error("Login failed");
    }
    const token = loginResult.data.access_token;
    localStorage.setItem(TOKEN_STORAGE_KEY, token);
    setAuthToken(token);

    const meResult = await apiClient.GET("/auth/me", {});
    if (!meResult.data) {
      throw new Error("Failed to fetch current user");
    }

    set({ token, user: meResult.data as AuthUser, isAuthenticated: true });
  },

  logout: () => {
    localStorage.removeItem(TOKEN_STORAGE_KEY);
    setAuthToken(null);
    set({ token: null, user: null, isAuthenticated: false });
  },
}));
```
