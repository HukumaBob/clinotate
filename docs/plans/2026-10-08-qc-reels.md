# QC Review Reels Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A "Review reels" panel on `/qc/:runId` that lists clips around doubtful moments (flicker plus a confidence grey zone), loops a chosen clip, and marks a clip reviewed when a flag lies inside it.

**Architecture:** Frontend only. A pure `buildReels` turns the stored predictions, the flicker events the screen already computes and a confidence band into clips; a small hook loops playback inside the active clip; a panel lists the clips. Frames are flagged with the existing buttons and F/N keys. No backend, endpoint or table changes.

**Tech Stack:** React, TypeScript, Vitest + Testing Library.

**Spec:** [docs/specs/2026-10-08-qc-reels-design.md](../specs/2026-10-08-qc-reels-design.md)

## Review Focus

- Flicker event whose `classId` is outside `doc.classes`: reason falls back to `class <id>`, no crash.
- A band with `low > high` is impossible: the panel clamps the other bound; an empty or NaN input is ignored.
- Clip padding never leaves `[0, durationMs]`; a moment at 0 ms or at the very end still gives a valid clip.
- Stride above 1: grey frames are only processed frames; a clip may contain few frames and still shows its count.
- Removing the only flag inside a clip turns it back to "not reviewed".
- The loop does nothing without a video element or without an active clip, and stops when the clip is deselected.
- Clicking the active clip again stops the loop instead of restarting it.

Ruling on spec ambiguity: "clips closer than 2 s merge" is measured between the **raw moments** (next start minus current end under 2000 ms), before padding; with 1 s padding per side such clips would overlap anyway.

// frontend/src/features/qc/reels.ts
import type { FlickerEvent } from "./analysis/flicker";
import type { PredictionsDocument } from "./predictions";

export interface GreyBand {
  low: number;
  high: number;
}

/** Detections the model is unsure about. Inclusive at both ends. */
export const DEFAULT_GREY_BAND: GreyBand = { low: 0.25, high: 0.5 };
export const REEL_PAD_MS = 1000;
export const REEL_MERGE_GAP_MS = 2000;

export interface Reel {
  startMs: number;
  endMs: number;
  reasons: string[];
  /** Processed frames inside the clip, by time. */
  frameIndices: number[];
}

interface Moment {
  startMs: number;
  endMs: number;
  reason: string;
}

export function buildReels(
  doc: PredictionsDocument,
  flickerEvents: FlickerEvent[],
  band: GreyBand,
  durationMs: number,
): Reel[] {
  const frames = Object.entries(doc.frames)
    .map(([key, frame]) => ({ index: Number(key), t: frame.timestamp_ms, detections: frame.detections }))
    .sort((a, b) => a.t - b.t);

  const moments: Moment[] = flickerEvents.map((event) => ({
    startMs: event.startMs,
    endMs: event.endMs,
    reason: `${event.kind}: ${doc.classes[event.classId] ?? `class ${event.classId}`}`,
  }));
  for (const frame of frames) {
    if (frame.detections.some((d) => d.confidence >= band.low && d.confidence <= band.high)) {
      moments.push({ startMs: frame.t, endMs: frame.t, reason: "low confidence" });
    }
  }
  moments.sort((a, b) => a.startMs - b.startMs);

  // Merge by the gap between raw moments, then pad.
  const groups: Moment[][] = [];
  let groupEnd = -Infinity;
  for (const moment of moments) {
    if (groups.length > 0 && moment.startMs - groupEnd < REEL_MERGE_GAP_MS) {
      groups[groups.length - 1].push(moment);
      groupEnd = Math.max(groupEnd, moment.endMs);
    } else {
      groups.push([moment]);
      groupEnd = moment.endMs;
    }
  }

  return groups.map((group) => {
    const startMs = Math.max(0, Math.min(...group.map((m) => m.startMs)) - REEL_PAD_MS);
    const endMs = Math.min(durationMs, Math.max(...group.map((m) => m.endMs)) + REEL_PAD_MS);
    return {
      startMs,
      endMs,
      reasons: [...new Set(group.map((m) => m.reason))],
      frameIndices: frames.filter((f) => f.t >= startMs && f.t <= endMs).map((f) => f.index),
    };
  });
}

export function isReviewed(reel: Reel, flags: { frame_index: number }[]): boolean {
  const inside = new Set(reel.frameIndices);
  return flags.some((flag) => inside.has(flag.frame_index));
}

// frontend/src/features/qc/useReelLoop.ts
import { useEffect } from "react";

import type { Reel } from "./reels";

/** Slack before the clip start, so a seek that lands a hair early is not fought. */
const BEFORE_START_SLACK_MS = 250;

/** Keeps the video inside `reel`: at the end, or when scrubbed out before the
 * start, it returns to the clip start. A rAF loop like PredictionOverlay's, so
 * playback, pausing and scrubbing share one mechanism. */
export function useReelLoop(
  videoRef: React.RefObject<HTMLVideoElement | null>,
  reel: Pick<Reel, "startMs" | "endMs"> | null,
): void {
  const startMs = reel?.startMs;
  const endMs = reel?.endMs;

  useEffect(() => {
    if (startMs === undefined || endMs === undefined) return;
    let handle = 0;

    function tick() {
      const el = videoRef.current;
      if (el) {
        const nowMs = el.currentTime * 1000;
        if (nowMs >= endMs! || nowMs < startMs! - BEFORE_START_SLACK_MS) {
          el.currentTime = startMs! / 1000;
        }
      }
      handle = requestAnimationFrame(tick);
    }

    handle = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(handle);
  }, [videoRef, startMs, endMs]);
}

// frontend/src/features/qc/ReelsPanel.tsx
import { formatClock } from "./analysis/strip";
import { isReviewed, type GreyBand, type Reel } from "./reels";

interface ReelsPanelProps {
  reels: Reel[];
  flags: { frame_index: number }[];
  activeIndex: number | null;
  band: GreyBand;
  onBandChange: (band: GreyBand) => void;
  onSelect: (index: number | null) => void;
  isCoarse: boolean;
}

const BUTTON =
  "rounded border border-gray-700 px-2 py-1 text-xs text-gray-300 hover:text-gray-100 disabled:opacity-50";
const FIELD = "w-16 rounded border border-gray-700 bg-gray-950 px-1 py-0.5 text-right text-xs text-gray-100";

export function ReelsPanel({ reels, flags, activeIndex, band, onBandChange, onSelect, isCoarse }: ReelsPanelProps) {
  const empty = reels.length === 0;
  const prev = activeIndex === null ? null : activeIndex - 1;
  const next = activeIndex === null ? 0 : activeIndex + 1;

  function changeBand(side: "low" | "high", raw: string) {
    if (raw.trim() === "") return;
    const value = Number(raw);
    if (!Number.isFinite(value)) return;
    const clamped = Math.min(Math.max(value, 0), 1);
    onBandChange(
      side === "low"
        ? { low: clamped, high: Math.max(band.high, clamped) }
        : { low: Math.min(band.low, clamped), high: clamped },
    );
  }

  return (
    <section aria-label="Review reels" className="flex flex-col gap-2">
      <h2 className="text-sm font-semibold text-gray-200">Review reels</h2>
      <div className="flex items-center gap-2 text-xs text-gray-400">
        <label className="flex items-center gap-1">
          Grey zone from
          <input
            type="number" min={0} max={1} step={0.05} className={FIELD}
            value={band.low}
            onChange={(e) => changeBand("low", e.target.value)}
          />
        </label>
        <label className="flex items-center gap-1">
          Grey zone to
          <input
            type="number" min={0} max={1} step={0.05} className={FIELD}
            value={band.high}
            onChange={(e) => changeBand("high", e.target.value)}
          />
        </label>
      </div>
      {isCoarse && (
        <p className="text-xs text-amber-300">
          The flicker window is coarse at this stride, so flicker clips may be misleading.
        </p>
      )}
      <div className="flex items-center gap-2">
        <button
          type="button" className={BUTTON}
          disabled={empty || prev === null || prev < 0}
          onClick={() => prev !== null && onSelect(prev)}
        >
          Previous reel
        </button>
        <button
          type="button" className={BUTTON}
          disabled={empty || next >= reels.length}
          onClick={() => onSelect(next)}
        >
          Next reel
        </button>
      </div>
      {empty ? (
        <p className="text-xs text-gray-500">No doubtful moments at these settings</p>
      ) : (
        <ul className="flex max-h-64 flex-col gap-1 overflow-y-auto">
          {reels.map((reel, index) => (
            <li key={`${reel.startMs}-${reel.endMs}`}>
              <button
                type="button"
                aria-pressed={index === activeIndex}
                onClick={() => onSelect(index === activeIndex ? null : index)}
                className={`flex w-full flex-col gap-1 rounded border px-2 py-1 text-left text-xs ${
                  index === activeIndex ? "border-blue-500 bg-gray-800" : "border-gray-800 hover:border-gray-600"
                }`}
              >
                <span className="text-gray-200">
                  {formatClock(reel.startMs)}–{formatClock(reel.endMs)}
                </span>
                <span className="flex flex-wrap items-center gap-1 text-gray-400">
                  {reel.reasons.map((reason) => (
                    <span key={reason} className="rounded bg-gray-800 px-1.5 py-0.5">{reason}</span>
                  ))}
                  <span>{reel.frameIndices.length === 1 ? "1 frame" : `${reel.frameIndices.length} frames`}</span>
                  {isReviewed(reel, flags) && <span className="text-green-400">Reviewed</span>}
